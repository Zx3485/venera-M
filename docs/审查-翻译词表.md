## 附：校验脚本

```python
# -*- coding: utf-8 -*-
"""校验 translation.json 的双侧一致性与重复 key。"""
import json, sys, collections

PATH = 'assets/translation.json'

def main():
    raw = open(PATH, encoding='utf-8').read()
    d = json.load(open(PATH, encoding='utf-8'),
                   object_pairs_hook=collections.OrderedDict)

    cn, tw = d['zh_CN'], d['zh_TW']

    print(f'zh_CN: {len(cn)} 条   zh_TW: {len(tw)} 条')

    # 1) 差集
    only_cn = sorted(set(cn) - set(tw))
    only_tw = sorted(set(tw) - set(cn))
    print(f'仅 zh_CN 有: {len(only_cn)}')
    for k in only_cn:
        print(f'   {k!r} -> {cn[k]!r}')
    print(f'仅 zh_TW 有: {len(only_tw)}')
    for k in only_tw:
        print(f'   {k!r} -> {tw[k]!r}')

    # 2) 重复 key（用普通 dict 解析，后者会覆盖前者）
    for seg in ('zh_CN', 'zh_TW'):
        pairs = json.loads(raw, object_pairs_hook=lambda p: p)
        # 手工统计每个 key 在本段出现的次数
        keys = [k for k, v in d_items(raw, seg)]
        dup = [k for k, c in collections.Counter(keys).items() if c > 1]
        print(f'{seg} 重复 key: {dup if dup else "无"}')

    # 3) 空值 / 占位符
    bad = [(s, k) for s, seg in (('zh_CN', cn), ('zh_TW', tw))
           for k, v in seg.items()
           if not v.strip() or v in ('TODO', 'xxx') or v == k]
    print(f'空值/占位符: {bad if bad else "无"}')

    return 0 if not (only_cn or only_tw) else 1

def d_items(raw, seg):
    """按段返回 (key, value) 序列，用于统计重复 key。"""
    data = json.loads(raw, object_pairs_hook=lambda p: p)
    for k, v in data:
        if k == seg:
            return v
    return []

if __name__ == '__main__':
    sys.exit(main())
```
